// Gemini APIを使ったAI解析 async function analyzeImage(base64Image) { const apiKey = apiKeyInput.value.trim(); if (!apiKey) { alert('Google Gemini API Keyを入力してください。'); loading.style.display = 'none'; return; } if (!base64Image) { alert('画像の切り出しに失敗しました。再度撮影してください。'); loading.style.display = 'none'; return; } const prompt = `この画像に写っている動物、昆虫、または植物を判別してください。 考えられる候補を可能性が高い順に最大3つまで挙げてください。 返答は必ず以下のJSON形式のみで出力してください（他の解説文は含めないでください）。 [ { "name": "生物の名前（和名）", "confidence": "高/中/低", "description": "判別した特徴や根拠の簡潔な説明（50文字程度）" } ]`; try { // モデル名を gemini-2.5-flash に変更 const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [ { text: prompt }, { inlineData: { mimeType: "image/jpeg", data: base64Image } } ] }] }) }); const data = await response.json(); // APIからのエラーチェック if (data.error) { throw new Error(`APIエラー (${data.error.code}): ${data.error.message}`); } if (!data.candidates || data.candidates.length === 0) { throw new Error("AIからの回答が得られませんでした。"); } const rawText = data.candidates[0].content.parts[0].text; // JSON部分のみ抽出 const jsonMatch = rawText.match(/\[[\s\S]*\]/); if (!jsonMatch) throw new Error("JSONのパースに失敗しました"); const candidates = JSON.parse(jsonMatch[0]); displayResults(candidates); } catch (err) { alert('解析エラーが発生しました: ' + err.message + '\n\n※APIキーが正しいか、Google AI Studioで再確認してください。'); } finally { loading.style.display = 'none'; } }
